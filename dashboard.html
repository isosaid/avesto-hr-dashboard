Нужна одна настройка базы

База пока разрешает только I и II квартал. Чтобы сохранять III и IV квартал, выполните этот SQL один раз:

ALTER TABLE courses DROP CONSTRAINT IF EXISTS courses_quarter_check;
ALTER TABLE courses ADD CONSTRAINT courses_quarter_check CHECK (quarter IN ('q1','q2','q3','q4'));
ALTER TABLE courses ADD COLUMN IF NOT EXISTS month_index INTEGER;
ALTER TABLE courses ADD COLUMN IF NOT EXISTS year INTEGER DEFAULT 2026;
